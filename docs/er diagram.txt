Table users {
  user_id int [pk, increment]
  name varchar(100)
  email varchar(150) [not null, unique]
  phone varchar(15)
  password_hash varchar(255) [not null]
  role varchar(20) [not null]
  account_status varchar(20) [not null]
  created_at datetime
}

Table donors {
  donor_id int [pk, increment]
  user_id int [not null, unique]
  blood_group varchar(5) [not null]
  address varchar(255)
  city varchar(100)
  state varchar(100)
  latitude decimal(10,7)
  longitude decimal(10,7)
  availability varchar(20) [not null]
  last_donation_date date
  donation_count int
}

Table blood_requests {
  request_id int [pk, increment]
  requester_id int [not null]
  blood_group_required varchar(5) [not null]
  units_required int [not null]
  hospital_name varchar(150) [not null]
  hospital_address varchar(255)
  latitude decimal(10,7)
  longitude decimal(10,7)
  urgency varchar(20) [not null]
  required_date date
  required_time time
  description text
  status varchar(30) [not null]
  created_at datetime
}

Table matches {
  match_id int [pk, increment]
  request_id int [not null]
  donor_id int [not null]
  distance_km decimal(8,2)
  match_score decimal(5,2)
  response_status varchar(20) [not null]
  notification_status varchar(20) [not null]
  matched_at datetime
  responded_at datetime
}

Table donations {
  donation_id int [pk, increment]
  donor_id int [not null]
  request_id int [not null]
  donation_date date
  units_donated int [not null]
  hospital_name varchar(150)
  status varchar(20) [not null]
}

Table notifications {
  notification_id int [pk, increment]
  user_id int [not null]
  request_id int [not null]
  title varchar(150) [not null]
  message text [not null]
  notification_type varchar(30)
  is_read boolean [not null]
  created_at datetime
}

Ref: donors.user_id > users.user_id

Ref: blood_requests.requester_id > users.user_id

Ref: matches.request_id > blood_requests.request_id

Ref: matches.donor_id > donors.donor_id

Ref: donations.donor_id > donors.donor_id

Ref: donations.request_id > blood_requests.request_id

Ref: notifications.user_id > users.user_id

Ref: notifications.request_id > blood_requests.request_id
